cmake_minimum_required(VERSION 3.20)
project(VLCUnityPortable LANGUAGES CXX)

option(VLCUNITY_BUILD_TESTS "Build host ABI/lifetime tests with a mock LibVLC" OFF)
option(VLCUNITY_TEST_ONLY "Only build the host tests (also usable on Windows)" OFF)
option(VLCUNITY_ENABLE_GPU "Build the platform GPU-sharing backend, with CPU callback fallback" ON)
option(VLCUNITY_BUILD_LINUX_INTEROP_SMOKE "Build a real EGL/OpenGL shared-texture driver test" OFF)
set(VLCUNITY_REQUIRED_CHANGESET "d94fd0473f" CACHE STRING "Pinned LibVLC 4 source changeset required by the bundled managed ABI")
set(VLCUNITY_LIBVLC_LIBRARIES "" CACHE STRING "iOS final-link decoder/static-plugin/contrib libraries (optional for archive generation)")

if(NOT VLCUNITY_TEST_ONLY)
    if(WIN32)
        message(FATAL_ERROR "Use build.ps1 for Windows graphics interop; this CMake target is the portable bridge.")
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
        add_library(VLCUnityPlugin STATIC PortableRenderingPlugin.cpp)
        target_compile_definitions(VLCUnityPlugin PRIVATE VLCUNITY_IOS=1 VLCUNITY_STATIC_LIBVLC=1)
        # Archive generation deliberately does not require LibVLC on the build
        # host. The complete matching native bundle is linked by the Xcode app.
        if(VLCUNITY_LIBVLC_LIBRARIES)
            target_link_libraries(VLCUnityPlugin PUBLIC ${VLCUNITY_LIBVLC_LIBRARIES})
        endif()
    else()
        add_library(VLCUnityPlugin SHARED PortableRenderingPlugin.cpp)
        target_link_libraries(VLCUnityPlugin PRIVATE ${CMAKE_DL_LIBS})
        if(APPLE)
            set_target_properties(VLCUnityPlugin PROPERTIES MACOSX_RPATH ON INSTALL_RPATH "@loader_path")
        else()
            target_link_options(VLCUnityPlugin PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/PortableExports.map")
            set_property(TARGET VLCUnityPlugin APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/PortableExports.map")
            if(NOT ANDROID)
                set_target_properties(VLCUnityPlugin PROPERTIES INSTALL_RPATH "$ORIGIN")
            endif()
        endif()
    endif()
    find_package(Threads REQUIRED)
    target_link_libraries(VLCUnityPlugin PRIVATE Threads::Threads)
    target_compile_features(VLCUnityPlugin PRIVATE cxx_std_17)
    target_compile_definitions(VLCUnityPlugin PRIVATE VLCUNITY_REQUIRED_CHANGESET="${VLCUNITY_REQUIRED_CHANGESET}")
    if(VLCUNITY_ENABLE_GPU)
        target_compile_definitions(VLCUnityPlugin PRIVATE VLCUNITY_ENABLE_GPU=1)
        target_include_directories(VLCUnityPlugin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty")
        if(ANDROID)
            target_sources(VLCUnityPlugin PRIVATE PortableAndroidInterop.cpp)
            target_link_libraries(VLCUnityPlugin PRIVATE EGL GLESv3)
        elseif(APPLE)
            enable_language(OBJCXX)
            target_sources(VLCUnityPlugin PRIVATE PortableAppleInterop.mm)
            set_target_properties(VLCUnityPlugin PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
            target_compile_options(VLCUnityPlugin PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
            target_link_libraries(VLCUnityPlugin PRIVATE "-framework Foundation" "-framework Metal" "-framework CoreVideo" "-framework IOSurface")
            if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
                target_link_libraries(VLCUnityPlugin PRIVATE "-framework OpenGLES" "-framework QuartzCore")
            else()
                target_link_libraries(VLCUnityPlugin PRIVATE "-framework OpenGL")
            endif()
        elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_sources(VLCUnityPlugin PRIVATE PortableLinuxInterop.cpp)
            find_package(OpenGL REQUIRED)
            find_path(VLCUNITY_EGL_INCLUDE_DIR EGL/egl.h REQUIRED)
            find_library(VLCUNITY_EGL_LIBRARY NAMES EGL libEGL.so.1 REQUIRED)
            find_package(X11 REQUIRED)
            target_include_directories(VLCUnityPlugin PRIVATE "${VLCUNITY_EGL_INCLUDE_DIR}")
            target_link_libraries(VLCUnityPlugin PRIVATE OpenGL::GL "${VLCUNITY_EGL_LIBRARY}" X11::X11)
        else()
            message(FATAL_ERROR "No GPU backend for this platform; configure VLCUNITY_ENABLE_GPU=OFF for CPU callbacks.")
        endif()
    endif()
    set_target_properties(VLCUnityPlugin PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE ON)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(VLCUnityPlugin PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    endif()
    if(ANDROID)
        target_sources(VLCUnityPlugin PRIVATE PortableAndroidJava.cpp)
        # Compatible with current Android 16 KiB page-size devices.
        # On Windows CMake may use an 8.3 path for clang++.exe whose short name
        # no longer ends in ++. Explicit driver mode retains C++ runtime linkage.
        target_link_options(VLCUnityPlugin PRIVATE "--driver-mode=g++" "-Wl,-z,max-page-size=16384")
    endif()
    install(TARGETS VLCUnityPlugin LIBRARY DESTINATION . ARCHIVE DESTINATION .)
endif()

if(VLCUNITY_BUILD_LINUX_INTEROP_SMOKE)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
        message(FATAL_ERROR "The Linux interop smoke test requires a native Linux build host.")
    endif()
    find_package(OpenGL REQUIRED)
    find_path(VLCUNITY_EGL_INCLUDE_DIR EGL/egl.h REQUIRED)
    find_library(VLCUNITY_EGL_LIBRARY NAMES EGL libEGL.so.1 REQUIRED)
    find_package(X11 REQUIRED)
    find_package(Threads REQUIRED)
    add_executable(PortableLinuxInteropSmoke PortableLinuxInterop.cpp PortableLinuxInteropSmoke.cpp)
    target_compile_features(PortableLinuxInteropSmoke PRIVATE cxx_std_17)
    target_include_directories(PortableLinuxInteropSmoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty" "${VLCUNITY_EGL_INCLUDE_DIR}")
    target_compile_options(PortableLinuxInteropSmoke PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
    target_link_libraries(PortableLinuxInteropSmoke PRIVATE OpenGL::GL "${VLCUNITY_EGL_LIBRARY}" X11::X11 Threads::Threads)
    if(VLCUNITY_BUILD_TESTS)
        add_executable(PortableLinuxVulkanHookTests PortableLinuxVulkanHookTests.cpp)
        target_compile_features(PortableLinuxVulkanHookTests PRIVATE cxx_std_17)
        target_include_directories(PortableLinuxVulkanHookTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty" "${VLCUNITY_EGL_INCLUDE_DIR}")
        target_compile_options(PortableLinuxVulkanHookTests PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
        target_link_libraries(PortableLinuxVulkanHookTests PRIVATE OpenGL::GL "${VLCUNITY_EGL_LIBRARY}" X11::X11 Threads::Threads)
        add_test(NAME PortableLinuxVulkanPreload COMMAND PortableLinuxVulkanHookTests)
    endif()
endif()

if(VLCUNITY_BUILD_TESTS OR VLCUNITY_TEST_ONLY)
    if(CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Portable ABI unit tests must run on the build host; use a separate host build tree.")
    endif()
    enable_testing()
    add_executable(PortableBridgeTests PortableRenderingPlugin.cpp PortableBridgeTests.cpp)
    target_compile_features(PortableBridgeTests PRIVATE cxx_std_17)
    target_compile_definitions(PortableBridgeTests PRIVATE VLCUNITY_STATIC_LIBVLC=1 VLCUNITY_REQUIRED_CHANGESET="d94fd0473f")
    if(MSVC)
        target_compile_options(PortableBridgeTests PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(PortableBridgeTests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME PortableBridgeAbiAndLifetime COMMAND PortableBridgeTests)

    # Link the real GPU-enabled bridge to a controlled backend/decoder. This
    # target needs neither platform GPU SDKs nor a native VLC distribution.
    find_package(Threads REQUIRED)
    add_executable(PortableGpuContextTests PortableRenderingPlugin.cpp PortableGpuContextTests.cpp)
    target_compile_features(PortableGpuContextTests PRIVATE cxx_std_17)
    target_compile_definitions(PortableGpuContextTests PRIVATE VLCUNITY_STATIC_LIBVLC=1 VLCUNITY_ENABLE_GPU=1 VLCUNITY_REQUIRED_CHANGESET="d94fd0473f")
    target_link_libraries(PortableGpuContextTests PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(PortableGpuContextTests PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(PortableGpuContextTests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME PortableGpuContextLifecycle COMMAND PortableGpuContextTests)
    set_tests_properties(PortableGpuContextLifecycle PROPERTIES TIMEOUT 20)
endif()
