cmake_minimum_required(VERSION 3.20)
project(MajVulkanSyncHook LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic supports Windows x64 only.")
endif()
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(MajVulkanSyncHook SHARED Hook.cpp)
target_compile_features(MajVulkanSyncHook PRIVATE cxx_std_17)
target_include_directories(MajVulkanSyncHook PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
set_target_properties(MajVulkanSyncHook PROPERTIES PREFIX "" OUTPUT_NAME "MajdataPlayVulkanPacing")
if(MSVC)
    target_compile_options(MajVulkanSyncHook PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET MajVulkanSyncHook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MajVulkanSyncHook PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(VulkanSyncHookStateTests CPUtests.cpp)
target_compile_features(VulkanSyncHookStateTests PRIVATE cxx_std_17)
target_include_directories(VulkanSyncHookStateTests PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
if(MSVC)
    target_compile_options(VulkanSyncHookStateTests PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET VulkanSyncHookStateTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(VulkanSyncHookStateTests PRIVATE -Wall -Wextra -Werror)
    target_link_options(VulkanSyncHookStateTests PRIVATE -static)
endif()
