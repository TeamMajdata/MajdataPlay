cmake_minimum_required(VERSION 3.20)
project(MajBufferBarrierProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic supports Windows x64 only.")
endif()
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(MajBufferBarrierProbe SHARED Probe.cpp)
target_compile_features(MajBufferBarrierProbe PRIVATE cxx_std_17)
target_include_directories(MajBufferBarrierProbe PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
set_target_properties(MajBufferBarrierProbe PROPERTIES PREFIX "" OUTPUT_NAME "MajdataPlayVulkanPacing")
if(MSVC)
    target_compile_options(MajBufferBarrierProbe PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET MajBufferBarrierProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MajBufferBarrierProbe PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(BufferBarrierStateTests CPUtests.cpp)
target_compile_features(BufferBarrierStateTests PRIVATE cxx_std_17)
target_include_directories(BufferBarrierStateTests PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
if(MSVC)
    target_compile_options(BufferBarrierStateTests PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET BufferBarrierStateTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(BufferBarrierStateTests PRIVATE -Wall -Wextra -Werror)
    target_link_options(BufferBarrierStateTests PRIVATE -static)
endif()
