cmake_minimum_required(VERSION 3.20)
project(MajBufferBarrierProbeV2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic supports Windows x64 only.")
endif()
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(MajBufferBarrierProbeV2 SHARED Probe.cpp)
target_compile_features(MajBufferBarrierProbeV2 PRIVATE cxx_std_17)
target_include_directories(MajBufferBarrierProbeV2 PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
set_target_properties(MajBufferBarrierProbeV2 PROPERTIES PREFIX "" OUTPUT_NAME "MajdataPlayVulkanPacing")
if(MSVC)
    target_compile_options(MajBufferBarrierProbeV2 PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET MajBufferBarrierProbeV2 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MajBufferBarrierProbeV2 PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(BufferBarrierV2StateTests CPUtests.cpp)
target_compile_features(BufferBarrierV2StateTests PRIVATE cxx_std_17)
target_include_directories(BufferBarrierV2StateTests PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
if(MSVC)
    target_compile_options(BufferBarrierV2StateTests PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET BufferBarrierV2StateTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(BufferBarrierV2StateTests PRIVATE -Wall -Wextra -Werror)
    target_link_options(BufferBarrierV2StateTests PRIVATE -static)
endif()
