cmake_minimum_required(VERSION 3.20)
project(MajQueueFenceProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This diagnostic supports Windows x64 only.")
endif()
get_filename_component(REPOSITORY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
add_library(MajQueueFenceProbe SHARED Marker.cpp)
target_compile_features(MajQueueFenceProbe PRIVATE cxx_std_17)
target_include_directories(MajQueueFenceProbe PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
set_target_properties(MajQueueFenceProbe PROPERTIES PREFIX "" OUTPUT_NAME "MajdataPlayVulkanPacing")
if(MSVC)
    target_compile_options(MajQueueFenceProbe PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET MajQueueFenceProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MajQueueFenceProbe PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(MarkerStateTests StateTests.cpp)
target_compile_features(MarkerStateTests PRIVATE cxx_std_17)
target_include_directories(MarkerStateTests PRIVATE
    "${REPOSITORY}/Tools/FFmpeg/Native/Unity"
    "${REPOSITORY}/Tools/FFmpeg/Native/ThirdParty/Vulkan-Headers/include")
if(MSVC)
    target_compile_options(MarkerStateTests PRIVATE /W4 /WX /EHsc /utf-8)
    set_property(TARGET MarkerStateTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(MarkerStateTests PRIVATE -Wall -Wextra -Werror)
    target_link_options(MarkerStateTests PRIVATE -static)
endif()
