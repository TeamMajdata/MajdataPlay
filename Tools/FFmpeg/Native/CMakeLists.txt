cmake_minimum_required(VERSION 3.20)
project(FFmpegUnityBridge LANGUAGES C CXX)

set(FFMPEG_ROOT "" CACHE PATH "Installed FFmpeg prefix (matching FFmpeg.AutoGen ABI)")
set(UNITY_PLUGIN_API "${CMAKE_CURRENT_SOURCE_DIR}/Unity" CACHE PATH "Unity native plug-in headers")
set(FFU_VULKAN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Vulkan-Headers/include" CACHE PATH "Vulkan headers matching the FFmpeg Vulkan Video build")
set(FFU_D3D12_HEADERS "" CACHE PATH "Optional verified D3D12 header overlay for older MinGW toolchains")
find_path(AVUTIL_INCLUDE_DIR libavutil/frame.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
find_library(AVUTIL_LIBRARY NAMES avutil libavutil avutil-61 HINTS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/bin" REQUIRED)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(BRIDGE_KIND STATIC)
else()
  set(BRIDGE_KIND SHARED)
endif()


add_library(FFmpegUnityBridge ${BRIDGE_KIND} Bridge.cpp)
target_compile_features(FFmpegUnityBridge PRIVATE cxx_std_17)
target_include_directories(FFmpegUnityBridge PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}")
target_link_libraries(FFmpegUnityBridge PRIVATE "${AVUTIL_LIBRARY}")
set_target_properties(FFmpegUnityBridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set_target_properties(FFmpegUnityBridge PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(WIN32)
  set(FFU_WINDOWS_SOURCES D3D11.cpp D3D12.cpp WglInterop.cpp VulkanInterop.cpp VulkanPortable.cpp VulkanVideoDecode.cpp)
  target_sources(FFmpegUnityBridge PRIVATE ${FFU_WINDOWS_SOURCES})
  target_include_directories(FFmpegUnityBridge PRIVATE "${FFU_VULKAN_HEADERS}")
  if(FFU_D3D12_HEADERS)
    target_include_directories(FFmpegUnityBridge BEFORE PRIVATE "${FFU_D3D12_HEADERS}")
  endif()
  target_link_libraries(FFmpegUnityBridge PRIVATE d3d11 d3d12 dxgi dxguid opengl32)
  target_compile_definitions(FFmpegUnityBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  # Unity's Windows plug-in importer expects no Unix "lib" prefix.
  set_target_properties(FFmpegUnityBridge PROPERTIES PREFIX "")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
  target_sources(FFmpegUnityBridge PRIVATE VulkanPortable.cpp VulkanPlatform.cpp VulkanVideoDecode.cpp)
  target_include_directories(FFmpegUnityBridge PRIVATE "${FFU_VULKAN_HEADERS}")
  if(ANDROID)
    find_library(AVCODEC_LIBRARY NAMES avcodec HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    target_sources(FFmpegUnityBridge PRIVATE AndroidMediaCodec.cpp)
    target_link_libraries(FFmpegUnityBridge PRIVATE "${AVCODEC_LIBRARY}" dl)
    target_link_options(FFmpegUnityBridge PRIVATE "-Wl,-z,max-page-size=16384")
  else()
    target_sources(FFmpegUnityBridge PRIVATE LinuxVaapi.cpp)
    target_link_options(FFmpegUnityBridge PRIVATE -static-libstdc++ -static-libgcc)
    set_target_properties(FFmpegUnityBridge PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH YES)
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(FFmpegUnityBridge PRIVATE Metal.mm)
  set_source_files_properties(Metal.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(FFmpegUnityBridge PRIVATE "-framework Metal" "-framework CoreVideo" "-framework Foundation")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_sources(FFmpegUnityBridge PRIVATE RegisterPlugin.mm)
    target_compile_definitions(FFmpegUnityBridge PRIVATE UnityPluginLoad=FfuUnityPluginLoad UnityPluginUnload=FfuUnityPluginUnload)
  else()
    set_target_properties(FFmpegUnityBridge PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
  endif()
endif()

if(MINGW)
  # Statically include the compiler's C++/thread runtimes, but keep FFmpeg's
  # explicitly resolved shared-library import archive dynamically linked.
  target_link_options(FFmpegUnityBridge PRIVATE -static)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_sources(FFmpegUnityBridge PRIVATE ExportsWin32MinGW.def)
  endif()
endif()
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_link_options(FFmpegUnityBridge PRIVATE
    "/EXPORT:UnityPluginLoad=_UnityPluginLoad@4" "/EXPORT:UnityPluginUnload=_UnityPluginUnload@0"
    "/EXPORT:UnityPluginLoad@4=_UnityPluginLoad@4" "/EXPORT:UnityPluginUnload@0=_UnityPluginUnload@0")
endif()

install(TARGETS FFmpegUnityBridge RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)

option(FFU_BUILD_TESTS "Build native D3D11 adapter smoke test" OFF)
option(FFU_BUILD_VULKAN_TESTS "Build Vulkan negotiation/lifetime and real GPU compute tests" OFF)
if(FFU_BUILD_VULKAN_TESTS AND (WIN32 OR (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID)))
  # This translation unit includes VulkanVideoDecode.cpp to exercise private
  # negotiation/lifetime policy while using real FFmpeg AVBufferRef ownership.
  add_executable(FFmpegUnityVulkanNegotiation tests/VulkanVideoNegotiation.cpp VulkanPortable.cpp)
  target_compile_features(FFmpegUnityVulkanNegotiation PRIVATE cxx_std_17)
  target_include_directories(FFmpegUnityVulkanNegotiation PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}" "${FFU_VULKAN_HEADERS}")
  target_link_libraries(FFmpegUnityVulkanNegotiation PRIVATE "${AVUTIL_LIBRARY}")
  if(WIN32)
    target_compile_definitions(FFmpegUnityVulkanNegotiation PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_target_properties(FFmpegUnityVulkanNegotiation PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
  if(MINGW)
    target_link_options(FFmpegUnityVulkanNegotiation PRIVATE -static)
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_library(FFU_TEST_VULKAN_LOADER NAMES vulkan libvulkan.so.1 REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(FFmpegUnityVulkanNegotiation PRIVATE Threads::Threads)
    add_executable(FFmpegUnityVulkanCompute tests/VulkanPortableSmoke.cpp VulkanPortable.cpp VulkanVideoDecode.cpp)
    target_compile_features(FFmpegUnityVulkanCompute PRIVATE cxx_std_17)
    target_include_directories(FFmpegUnityVulkanCompute PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}" "${FFU_VULKAN_HEADERS}")
    target_link_libraries(FFmpegUnityVulkanCompute PRIVATE "${AVUTIL_LIBRARY}" "${FFU_TEST_VULKAN_LOADER}" Threads::Threads)
  endif()
endif()
if(WIN32 AND FFU_BUILD_TESTS)
  add_executable(FFmpegUnityBridgeSmoke tests/D3D11Smoke.cpp Bridge.cpp ${FFU_WINDOWS_SOURCES})
  target_compile_features(FFmpegUnityBridgeSmoke PRIVATE cxx_std_17)
  target_include_directories(FFmpegUnityBridgeSmoke PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}" "${FFU_VULKAN_HEADERS}")
  if(FFU_D3D12_HEADERS)
    target_include_directories(FFmpegUnityBridgeSmoke BEFORE PRIVATE "${FFU_D3D12_HEADERS}")
  endif()
  target_compile_definitions(FFmpegUnityBridgeSmoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(FFmpegUnityBridgeSmoke PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(FFmpegUnityBridgeSmoke PRIVATE "${AVUTIL_LIBRARY}" d3d11 d3d12 dxgi dxguid opengl32)
  if(MINGW)
    target_link_options(FFmpegUnityBridgeSmoke PRIVATE -static)
  endif()
endif()

if(WIN32 AND FFU_BUILD_TESTS)
  find_library(FFU_TEST_AVCODEC NAMES avcodec libavcodec avcodec-63 HINTS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/bin" REQUIRED)
  find_library(FFU_TEST_AVFORMAT NAMES avformat libavformat avformat-63 HINTS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/bin" REQUIRED)
  find_library(FFU_TEST_SWSCALE NAMES swscale libswscale swscale-10 HINTS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/bin" REQUIRED)
  add_executable(FFmpegD3D12VideoSmoke tests/D3D12VideoSmoke.cpp D3D12.cpp)
  target_compile_features(FFmpegD3D12VideoSmoke PRIVATE cxx_std_17)
  target_include_directories(FFmpegD3D12VideoSmoke PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}")
  if(FFU_D3D12_HEADERS)
    target_include_directories(FFmpegD3D12VideoSmoke BEFORE PRIVATE "${FFU_D3D12_HEADERS}")
  endif()
  target_compile_definitions(FFmpegD3D12VideoSmoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(FFmpegD3D12VideoSmoke PRIVATE "${AVUTIL_LIBRARY}" "${FFU_TEST_AVCODEC}" "${FFU_TEST_AVFORMAT}" "${FFU_TEST_SWSCALE}" d3d11 d3d12 dxgi dxguid)
  set_target_properties(FFmpegD3D12VideoSmoke PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MINGW)
    target_link_options(FFmpegD3D12VideoSmoke PRIVATE -static)
  endif()
endif()
