cmake_minimum_required(VERSION 3.20)
project(FFmpegUnityBridge LANGUAGES C CXX)

set(FFMPEG_ROOT "" CACHE PATH "Installed FFmpeg prefix (matching FFmpeg.AutoGen ABI)")
set(UNITY_PLUGIN_API "${CMAKE_CURRENT_SOURCE_DIR}/Unity" CACHE PATH "Unity native plug-in headers")
find_path(AVUTIL_INCLUDE_DIR libavutil/frame.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
find_library(AVUTIL_LIBRARY NAMES avutil libavutil avutil-61 HINTS "${FFMPEG_ROOT}/lib" "${FFMPEG_ROOT}/bin" REQUIRED)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(BRIDGE_KIND STATIC)
else()
  set(BRIDGE_KIND SHARED)
endif()

add_library(FFmpegUnityBridge ${BRIDGE_KIND} Bridge.cpp)
target_compile_features(FFmpegUnityBridge PRIVATE cxx_std_17)
target_include_directories(FFmpegUnityBridge PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}")
target_link_libraries(FFmpegUnityBridge PRIVATE "${AVUTIL_LIBRARY}")
set_target_properties(FFmpegUnityBridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
set_target_properties(FFmpegUnityBridge PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(WIN32)
  set(FFU_WINDOWS_SOURCES D3D11.cpp D3D12.cpp WglInterop.cpp VulkanInterop.cpp)
  target_sources(FFmpegUnityBridge PRIVATE ${FFU_WINDOWS_SOURCES})
  target_include_directories(FFmpegUnityBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Vulkan-Headers/include")
  target_link_libraries(FFmpegUnityBridge PRIVATE d3d11 d3d12 dxgi dxguid opengl32)
  target_compile_definitions(FFmpegUnityBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  # Unity's Windows plug-in importer expects no Unix "lib" prefix.
  set_target_properties(FFmpegUnityBridge PROPERTIES PREFIX "")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
  target_sources(FFmpegUnityBridge PRIVATE VulkanPortable.cpp VulkanPlatform.cpp)
  target_include_directories(FFmpegUnityBridge PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Vulkan-Headers/include")
  if(ANDROID)
    find_library(AVCODEC_LIBRARY NAMES avcodec HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
    target_sources(FFmpegUnityBridge PRIVATE AndroidMediaCodec.cpp)
    target_link_libraries(FFmpegUnityBridge PRIVATE "${AVCODEC_LIBRARY}" dl)
    target_link_options(FFmpegUnityBridge PRIVATE "-Wl,-z,max-page-size=16384")
  else()
    target_sources(FFmpegUnityBridge PRIVATE LinuxVaapi.cpp)
    target_link_options(FFmpegUnityBridge PRIVATE -static-libstdc++ -static-libgcc)
    set_target_properties(FFmpegUnityBridge PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH YES)
  endif()
elseif(APPLE)
  enable_language(OBJCXX)
  target_sources(FFmpegUnityBridge PRIVATE Metal.mm)
  set_source_files_properties(Metal.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(FFmpegUnityBridge PRIVATE "-framework Metal" "-framework CoreVideo" "-framework Foundation")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_sources(FFmpegUnityBridge PRIVATE RegisterPlugin.mm)
    target_compile_definitions(FFmpegUnityBridge PRIVATE UnityPluginLoad=FfuUnityPluginLoad UnityPluginUnload=FfuUnityPluginUnload)
  else()
    set_target_properties(FFmpegUnityBridge PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
  endif()
endif()

if(MINGW)
  # Statically include the compiler's C++/thread runtimes, but keep FFmpeg's
  # explicitly resolved shared-library import archive dynamically linked.
  target_link_options(FFmpegUnityBridge PRIVATE -static)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_sources(FFmpegUnityBridge PRIVATE ExportsWin32MinGW.def)
  endif()
endif()
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 4)
  target_link_options(FFmpegUnityBridge PRIVATE
    "/EXPORT:UnityPluginLoad=_UnityPluginLoad@4" "/EXPORT:UnityPluginUnload=_UnityPluginUnload@0"
    "/EXPORT:UnityPluginLoad@4=_UnityPluginLoad@4" "/EXPORT:UnityPluginUnload@0=_UnityPluginUnload@0")
endif()

install(TARGETS FFmpegUnityBridge RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)

option(FFU_BUILD_TESTS "Build native D3D11 adapter smoke test" OFF)
if(WIN32 AND FFU_BUILD_TESTS)
  add_executable(FFmpegUnityBridgeSmoke tests/D3D11Smoke.cpp Bridge.cpp ${FFU_WINDOWS_SOURCES})
  target_compile_features(FFmpegUnityBridgeSmoke PRIVATE cxx_std_17)
  target_include_directories(FFmpegUnityBridgeSmoke PRIVATE "${UNITY_PLUGIN_API}" "${AVUTIL_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/Vulkan-Headers/include")
  target_compile_definitions(FFmpegUnityBridgeSmoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(FFmpegUnityBridgeSmoke PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(FFmpegUnityBridgeSmoke PRIVATE "${AVUTIL_LIBRARY}" d3d11 d3d12 dxgi dxguid opengl32)
  if(MINGW)
    target_link_options(FFmpegUnityBridgeSmoke PRIVATE -static)
  endif()
endif()
