using System;

#nullable enable

namespace __ROOT_NAMESPACE__.Runtime
{
    /// <summary>
    /// Reports a Java throwable after clearing the current thread's pending JVM exception.
    /// </summary>
    /// <remarks>
    /// The message includes the original Java stack trace when diagnostics succeed.
    /// Unity class resolution may separately throw UnityEngine.AndroidJavaException.
    /// </remarks>
    public partial class JavaInvocationException : Exception
    {
        /// <summary>
        /// Gets the original Java stack trace, or an empty string if diagnostics failed.
        /// </summary>
        public string JavaStackTrace { get; }

        /// <summary>
        /// Initializes an invocation failure with the Java description and stack trace.
        /// </summary>
        /// <param name="message">The original Java throwable's description.</param>
        /// <param name="javaStackTrace">The recovered Java stack trace, or an empty string.</param>
        /// <exception cref="ArgumentNullException">Either diagnostic string is null.</exception>
        public JavaInvocationException(string message, string javaStackTrace)
            : base((message ?? throw new ArgumentNullException(nameof(message)))
                + Environment.NewLine + (javaStackTrace ?? throw new ArgumentNullException(nameof(javaStackTrace))))
        {
            JavaStackTrace = javaStackTrace;
        }
    }
}
